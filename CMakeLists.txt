cmake_minimum_required(VERSION 3.16)
project(psbbv_network_policy C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(psbbv_network_policy src/url_policy.c)
target_include_directories(psbbv_network_policy PUBLIC include)
target_compile_options(psbbv_network_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(test_url_policy tests/test_url_policy.c)
target_link_libraries(test_url_policy PRIVATE psbbv_network_policy)
target_compile_options(test_url_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME url_policy COMMAND test_url_policy)

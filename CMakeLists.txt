cmake_minimum_required(VERSION 3.16)
project(psbbv_network_policy C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(psbbv_endpoint_policy src/endpoint_policy.c)
target_include_directories(psbbv_endpoint_policy PUBLIC include)
target_compile_options(psbbv_endpoint_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(test_endpoint_policy tests/test_endpoint_policy.c)
target_link_libraries(test_endpoint_policy PRIVATE psbbv_endpoint_policy)
target_compile_options(test_endpoint_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME endpoint_policy COMMAND test_endpoint_policy)
